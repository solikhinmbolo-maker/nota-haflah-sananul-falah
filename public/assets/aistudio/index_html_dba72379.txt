      function angkaTerbilang(angka) {
        let bilangan = ["", "Satu", "Dua", "Tiga", "Empat", "Lima", "Enam", "Tujuh", "Delapan", "Sembilan", "Sepuluh", "Sebelas"];
        let bagi = function(a, b) { return Math.floor(a / b); };
        if (angka < 12) return bilangan[angka];
        if (angka < 20) return angkaTerbilang(angka - 10) + " Belas";
        if (angka < 100) return angkaTerbilang(bagi(angka, 10)) + " Puluh " + (angka % 10 > 0 ? angkaTerbilang(angka % 10) : "");
        if (angka < 200) return "Seratus " + (angka - 100 > 0 ? angkaTerbilang(angka - 100) : "");
        if (angka < 1000) return angkaTerbilang(bagi(angka, 100)) + " Ratus " + (angka % 100 > 0 ? angkaTerbilang(angka % 100) : "");
        if (angka < 2000) return "Seribu " + (angka - 1000 > 0 ? angkaTerbilang(angka - 1000) : "");
        if (angka < 1000000) return angkaTerbilang(bagi(angka, 1000)) + " Ribu " + (angka % 1000 > 0 ? angkaTerbilang(angka % 1000) : "");
        if (angka < 1000000000) return angkaTerbilang(bagi(angka, 1000000)) + " Juta " + (angka % 1000000 > 0 ? angkaTerbilang(angka % 1000000) : "");
        return "";
      }

      google.script.run.withSuccessHandler(classes => {
        let sel = document.getElementById('kelas');
        sel.innerHTML = '<option value="">-- Pilih Kelas Santri --</option>';
        classes.forEach(k => { sel.innerHTML += `<option value="${k}">${k}</option>`; });
      }).getClasses();

      function loadStudents() {
        let kelas = document.getElementById('kelas').value;
        let santriSelect = document.getElementById('santri');
        checkReady();
        if(kelas) {
          santriSelect.innerHTML = '<option value="">Mencari Data di Sheets...</option>';
          google.script.run.withSuccessHandler(students => {
            santriSelect.innerHTML = '<option value="">-- Pilih Nama Santri --</option>';
            students.forEach(s => {
              santriSelect.innerHTML += `<option value="${s.row}">${s.nama} (Rp ${s.nominal.toLocaleString('id-ID')})</option>`;
            });
          }).getStudentsByClass(kelas);
        } else {
          santriSelect.innerHTML = '<option value="">Pilih Kelas Dahulu</option>';
        }
      }

      function checkReady() {
        let row = document.getElementById('santri').value;
        let btn = document.getElementById('btnCetak');
        if(row) { 
          btn.classList.add("active"); 
          btn.innerHTML = "<i class='fas fa-eye'></i> Tampilkan Kuitansi"; 
          btn.disabled = false; 
        } else { 
          btn.classList.remove("active"); 
          btn.innerHTML = "<i class='fas fa-file-invoice'></i> Pilih Santri Dahulu"; 
          btn.disabled = true; 
        }
      }

      function bukaPreview() {
        let row = document.getElementById('santri').value;
        let btn = document.getElementById('btnCetak');
        btn.innerHTML = "<i class='fas fa-spinner fa-spin'></i> Memuat Dokumen..."; 
        btn.disabled = true;
        
        google.script.run.withSuccessHandler(data => {
          document.getElementById('pNoNota').innerText = data.noNota;
          document.getElementById('pTanggal').innerText = data.tanggal;
          document.getElementById('pNama').innerText = data.namaSantri;
          document.getElementById('pKelas').innerText = data.kelas;
          
          let nominalAngkaStr = data.nominalFormatted.replace(/[^0-9]/g, '');
          let nominalAngka = parseInt(nominalAngkaStr) || 0;
          
          let teksTerbilang = (nominalAngka > 0) ? angkaTerbilang(nominalAngka) + " RUPIAH" : "NOL RUPIAH";
          document.getElementById('pTerbilang').innerText = teksTerbilang.toUpperCase();
          document.getElementById('pNominal').innerText = data.nominalFormatted.replace("Rp ", "");
          
          if (data.keterangan && data.keterangan.toString().toUpperCase().trim() === "LUNAS") {
            document.getElementById('watermarkLunas').style.display = 'block';
          } else {
            document.getElementById('watermarkLunas').style.display = 'none';
          }
          
          document.getElementById('modal').style.display = 'flex';
          btn.innerHTML = "<i class='fas fa-eye'></i> Tampilkan Kuitansi"; 
          btn.disabled = false;
        }).getDetailNotaFast(row);
      }

      function tutupPreview() { 
        document.getElementById('modal').style.display = 'none'; 
      }

      function unduhJPEG() {
        let btnModal = document.getElementById('btnUnduhModal');
        let teksAsli = btnModal.innerHTML;
        btnModal.innerHTML = "<i class='fas fa-spinner fa-spin'></i> Menyimpan..."; 
        btnModal.disabled = true;

        html2canvas(document.getElementById('areaCetak'), { 
          scale: 3, 
          backgroundColor: "#ffffff", 
          useCORS: true,
          allowTaint: false,
          width: 1150,
          height: 540
        }).then(canvas => {
          let nama = document.getElementById('pNama').innerText;
          let link = document.createElement('a');
          link.download = `Kuitansi_Haflah_${nama}.jpeg`;
          link.href = canvas.toDataURL('image/jpeg', 1.0);
          link.click();
          btnModal.innerHTML = teksAsli; 
          btnModal.disabled = false;
        }).catch(err => {
          alert("Gagal memproses gambar. Muat ulang halaman lalu coba lagi.");
          btnModal.innerHTML = teksAsli; 
          btnModal.disabled = false;
        });
      }
